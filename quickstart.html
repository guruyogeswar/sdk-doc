Quickstart

Get a LIMBIC agent thinking in five minutes. You need a C++17 compiler, CMake 3.16+, and OpenMP 4.5 at runtime.

Platforms Linux x86_64 ships as a prebuilt tarball. macOS and Windows are supported by building from source - the codebase includes a Windows compatibility shim and a MinGW-w64 cross-compile path. See the SDK README for the build-from-source steps.

1. Install the SDK

Download the release archive for your platform and extract it:

shell bash# Linux x86_64 - prebuilt
tar xzf limbic-sdk-0.1.0-linux-x86_64.tar.gz
cd limbic-sdk-0.1.0-linux-x86_64
ls
# include/  lib/  examples/  README.md  LICENSE

On macOS or Windows, clone the repo and run the build script - the output layout matches the prebuilt tarball.

2. Link from CMake

CMakeLists.txt cmakecmake_minimum_required(VERSION 3.16)
project(my_game LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)

find_package(Limbic REQUIRED)

add_executable(my_game main.cpp)
target_link_libraries(my_game PRIVATE Limbic::limbic)
shell bashcmake -B build -DCMAKE_PREFIX_PATH=/path/to/limbic-sdk-0.1.0-linux-x86_64
cmake --build build

3. Hello, agent

main.cpp C++#include "limbic/limbic_engine.hpp"
#include <cstdio>
using namespace limbic::sdk;

int main() {
    EngineConfig cfg;
    cfg.preset     = Preset::COGNITIVE;
    cfg.max_agents = 10;
    cfg.seed       = 42;
    auto engine = LimbicEngine::create(cfg);

    engine.register_action("forage", TAG_RESOURCE | TAG_SURVIVAL);
    engine.register_action("trade",  TAG_SOCIAL   | TAG_COOPERATIVE);
    engine.register_action("rest",   TAG_ROUTINE  | TAG_SURVIVAL);

    AgentConfig ac{ .x=0, .y=0, .personality=Personality::MERCHANT };
    uint32_t npc = engine.spawn(ac);
    engine.warmup(npc, 100);

    for (int i = 0; i < 10; ++i) {
        float energy = 0.6f, hunger = 0.4f;
        engine.perceive(npc, Percept::ENERGY_LEVEL, &energy, 1);
        engine.perceive(npc, Percept::HUNGER,       &hunger, 1);
        engine.tick(0.016f);

        Decision d = engine.get_decision(npc);
        printf("tick %d: %-8s confidence=%.2f\n",
               i, d.action_name, d.confidence);

        engine.report_outcome(npc, 0.3f, 2.f);
    }
}

4. Run it

shell bash# Linux
LD_LIBRARY_PATH=/path/to/limbic-sdk/lib ./build/my_game

# macOS
DYLD_LIBRARY_PATH=/path/to/limbic-sdk/lib ./build/my_game

# Windows - place limbic.dll next to the .exe, then run it.

# tick 0: forage   confidence=0.62
# tick 1: forage   confidence=0.71
# tick 2: trade    confidence=0.58
# ...
Tip warmup() runs the brain forward without advancing the world clock, so freshly-spawned agents start with established neuromodulator baselines instead of a flat blank-slate.
